```dart
class CommonMistake {
  const CommonMistake({
    required this.mistake,
    required this.why,
    required this.fix,
  });

  factory CommonMistake.fromJson(Map<String, dynamic> json) => CommonMistake(
        mistake: json['mistake'] as String,
        why: json['why'] as String,
        fix: json['fix'] as String,
      );

  final String mistake;
  final String why;
  final String fix;

  Map<String, dynamic> toJson() => {'mistake': mistake, 'why': why, 'fix': fix};
}

class Exercise {
  const Exercise({
    required this.id,
    required this.nameTr,
    required this.nameEn,
    required this.category,
    required this.location,
    required this.equipment,
    required this.primaryMuscles,
    required this.secondaryMuscles,
    required this.difficulty,
    required this.summary,
    required this.setup,
    required this.execution,
    required this.breathing,
    required this.tempo,
    required this.cues,
    required this.commonMistakes,
    required this.safety,
    required this.regressions,
    required this.progressions,
    required this.isUserDefined,
    this.imagePath,
    this.videoQuery,
  });

  factory Exercise.fromJson(Map<String, dynamic> json) => Exercise(
        id: json['id'] as String,
        nameTr: json['nameTr'] as String,
        nameEn: json['nameEn'] as String,
        category: ExerciseCategory.values.byName(json['category'] as String),
        location: ExerciseLocation.values.byName(json['location'] as String),
        equipment: _strings(json['equipment']),
        primaryMuscles: _strings(json['primaryMuscles']),
        secondaryMuscles: _strings(json['secondaryMuscles']),
        difficulty: json['difficulty'] as int,
        summary: json['summary'] as String,
        setup: _strings(json['setup']),
        execution: _strings(json['execution']),
        breathing: json['breathing'] as String,
        tempo: json['tempo'] as String,
        cues: _strings(json['cues']),
        commonMistakes: (json['commonMistakes'] as List)
            .map((m) => CommonMistake.fromJson(m as Map<String, dynamic>))
            .toList(),
        safety: json['safety'] as String,
        regressions: _strings(json['regressions']),
        progressions: _strings(json['progressions']),
        imagePath: json['imagePath'] as String?,
        videoQuery: json['videoQuery'] as String?,
        isUserDefined: json['isUserDefined'] as bool? ?? false,
      );

  final String id;
  final String nameTr;
  final String nameEn;
  final ExerciseCategory category;
  final ExerciseLocation location;
  final List<String> equipment;
  final List<String> primaryMuscles;
  final List<String> secondaryMuscles;
  final int difficulty;
  final String summary;
  final List<String> setup;
  final List<String> execution;
  final String breathing;
  final String tempo;
  final List<String> cues;
  final List<CommonMistake> commonMistakes;
  final String safety;
  final List<String> regressions;
  final List<String> progressions;
  final String? imagePath;
  final String? videoQuery;
  final bool isUserDefined;

  Map<String, dynamic> toJson() => {
        'id': id,
        'nameTr': nameTr,
        'nameEn': nameEn,
        'category': category.name,
        'location': location.name,
        'equipment': equipment,
        'primaryMuscles': primaryMuscles,
        'secondaryMuscles': secondaryMuscles,
        'difficulty': difficulty,
        'summary': summary,
        'setup': setup,
        'execution': execution,
        'breathing': breathing,
        'tempo': tempo,
        'cues': cues,
        'commonMistakes': [for (final m in commonMistakes) m.toJson()],
        'safety': safety,
        'regressions': regressions,
        'progressions': progressions,
        if (imagePath != null) 'imagePath': imagePath,
        if (videoQuery != null) 'videoQuery': videoQuery,
        'isUserDefined': isUserDefined,
      };

  static List<String> _strings(Object? v) =>
      (v as List? ?? const []).cast<String>();
}
```

`app/lib/features/catalog/data/catalog_repository.dart`:

```dart
import 'dart:convert';

import 'package:drift/drift.dart';

import 'package:disport/core/db/app_database.dart';
import 'package:disport/features/catalog/domain/exercise.dart';

class CatalogRepository {
  CatalogRepository(this._db);

  final AppDatabase _db;

  Future<int> countAll() async {
    final c = _db.exercises.id.count();
    final q = _db.selectOnly(_db.exercises)..addColumns([c]);
    return (await q.getSingle()).read(c)!;
  }

  /// Tohum verisini yükler. Tablo doluysa hiçbir şey yapmaz.
  Future<void> seedFromJson(String jsonString) async {
    if (await countAll() > 0) return;
    final decoded = jsonDecode(jsonString) as Map<String, dynamic>;
    final items = (decoded['exercises'] as List)
        .map((e) => Exercise.fromJson(e as Map<String, dynamic>));
    await _db.batch((batch) {
      batch.insertAll(_db.exercises, [for (final e in items) _toRow(e)]);
    });
  }

  Stream<List<Exercise>> watchFiltered({
    String? query,
    ExerciseLocation? location,
    String? muscle,
  }) {
    final q = _db.select(_db.exercises)
      ..where((t) => t.deletedAt.isNull())
      ..orderBy([(t) => OrderingTerm.asc(t.nameTr)]);
    if (query != null && query.isNotEmpty) {
      final needle = '%${query.toLowerCase()}%';
      q.where((t) =>
          t.nameTr.lower().like(needle) | t.nameEn.lower().like(needle));
    }
    if (location != null) {
      q.where((t) =>
          t.location.equals(location.name) |
          t.location.equals(ExerciseLocation.both.name));
    }
    if (muscle != null) {
      q.where((t) => t.primaryMusclesJson.like('%"$muscle"%'));
    }
    return q.watch().map((rows) => [for (final r in rows) _fromRow(r)]);
  }

  Future<Exercise?> getById(String id) async {
    final row = await (_db.select(_db.exercises)
          ..where((t) => t.id.equals(id)))
        .getSingleOrNull();
    return row == null ? null : _fromRow(row);
  }

  Future<void> addUserDefined(Exercise e) =>
      _db.into(_db.exercises).insert(_toRow(e));

  ExercisesCompanion _toRow(Exercise e) => ExercisesCompanion.insert(
        id: e.id,
        updatedAt: DateTime.now().millisecondsSinceEpoch,
        nameTr: e.nameTr,
        nameEn: e.nameEn,
        category: e.category.name,
        location: e.location.name,
        difficulty: e.difficulty,
        isUserDefined: Value(e.isUserDefined),
        equipmentJson: jsonEncode(e.equipment),
        primaryMusclesJson: jsonEncode(e.primaryMuscles),
        detailJson: jsonEncode(e.toJson()),
      );

  Exercise _fromRow(Exercise_ row) => Exercise.fromJson(
      jsonDecode(row.detailJson) as Map<String, dynamic>);
}
```


ve `_fromRow(ExerciseRow row)` kullan. build_runner'ı yeniden çalıştır.

```dart
@riverpod
CatalogRepository catalogRepository(Ref ref) =>
    CatalogRepository(ref.watch(appDatabaseProvider));

/// Arama/filtre durumu.
@riverpod
class CatalogFilter extends _$CatalogFilter {
  @override
  ({String query, ExerciseLocation? location}) build() =>
      (query: '', location: null);

  void setQuery(String q) => state = (query: q, location: state.location);
  void setLocation(ExerciseLocation? l) =>
      state = (query: state.query, location: l);
}

@riverpod
Stream<List<Exercise>> filteredExercises(Ref ref) {
  final f = ref.watch(catalogFilterProvider);
  return ref.watch(catalogRepositoryProvider).watchFiltered(
        query: f.query,
        location: f.location,
      );
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import 'package:disport/features/catalog/application/catalog_providers.dart';
import 'package:disport/features/catalog/domain/exercise.dart';

class ExerciseDetailScreen extends ConsumerWidget {
  const ExerciseDetailScreen({super.key, required this.exerciseId});

  final String exerciseId;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final repo = ref.watch(catalogRepositoryProvider);

    return FutureBuilder<Exercise?>(
      future: repo.getById(exerciseId),
      builder: (context, snapshot) {
        final e = snapshot.data;
        if (e == null) {
          return const Scaffold(
              body: Center(child: CircularProgressIndicator()));
        }
        return DefaultTabController(
          length: 4,
          child: Scaffold(
            appBar: AppBar(
              title: Text(e.nameTr),
              bottom: const TabBar(isScrollable: true, tabs: [
                Tab(text: 'Nasıl yapılır'),
                Tab(text: 'Sık hatalar'),
                Tab(text: 'Kolaylaştır / Zorlaştır'),
                Tab(text: 'Güvenlik'),
              ]),
            ),
            body: TabBarView(children: [
              _HowTo(e: e),
              _Mistakes(e: e),
              _Variants(e: e),
              _Safety(e: e),
            ]),
          ),
        );
      },
    );
  }
}

class _HowTo extends StatelessWidget {
  const _HowTo({required this.e});
  final Exercise e;

  @override
  Widget build(BuildContext context) {
    return ListView(padding: const EdgeInsets.all(16), children: [
      if (e.imagePath != null) Image.asset(e.imagePath!, height: 180),
      Text(e.summary, style: Theme.of(context).textTheme.bodyLarge),
      const SizedBox(height: 12),
      Text('Başlangıç', style: Theme.of(context).textTheme.titleMedium),
      for (final s in e.setup) Text('• $s'),
      const SizedBox(height: 12),
      Text('Hareket', style: Theme.of(context).textTheme.titleMedium),
      for (final (i, s) in e.execution.indexed) Text('${i + 1}. $s'),
      const SizedBox(height: 12),
      Text('Nefes: ${e.breathing}'),
      Text('Tempo: ${e.tempo}'),
    ]);
  }
}

class _Mistakes extends StatelessWidget {
  const _Mistakes({required this.e});
  final Exercise e;

  @override
  Widget build(BuildContext context) {
    return ListView(padding: const EdgeInsets.all(16), children: [
      for (final m in e.commonMistakes)
        Card(
          child: Padding(
            padding: const EdgeInsets.all(12),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text('✗ ${m.mistake}',
                    style: Theme.of(context).textTheme.titleSmall),
                Text('Neden: ${m.why}'),
                Text('Düzeltme: ${m.fix}'),
              ],
            ),
          ),
        ),
    ]);
  }
}

class _Variants extends StatelessWidget {
  const _Variants({required this.e});
  final Exercise e;

  @override
  Widget build(BuildContext context) {
    return ListView(padding: const EdgeInsets.all(16), children: [
      Text('Kolaylaştır', style: Theme.of(context).textTheme.titleMedium),
      if (e.regressions.isEmpty) const Text('—'),
      for (final r in e.regressions) Text('• $r'),
      const SizedBox(height: 16),
      Text('Zorlaştır', style: Theme.of(context).textTheme.titleMedium),
      if (e.progressions.isEmpty) const Text('—'),
      for (final p in e.progressions) Text('• $p'),
    ]);
  }
}

class _Safety extends StatelessWidget {
  const _Safety({required this.e});

  @override
  Widget build(BuildContext context) {

    );
  }
```
